# 基本数据类型

执行下面的代码，定义数值类型变量。

```javascript
let num = 233;
console.log(num, typeof num);
let bin = 0b10101;
console.log(bin);
let oct = 0o777;
console.log(oct);
let hex = 0xff;
console.log(hex);
```

执行下面的代码，定义字符串类型变量。

```javascript
let str = 'Hello World!';
console.log(str, typeof str);
```

执行下面的代码，定义布尔类型变量。

```javascript
let bool = true;
console.log(bool, typeof bool);
```

执行下面的代码，定义未定义类型变量。

```javascript
let undef;
console.log(undef, typeof undef);
```

执行下面的代码，定义空类型变量。

```javascript
let NULL = null;
console.log(NULL, typeof NULL);
```

执行下面的代码，定义比数值类型支持范围更大的数值类型变量。

```javascript
let num1 = 233n;
console.log(num1, typeof num1);
let num2 = BigInt(666);
console.log(num2, typeof num2);
let max = Number.MAX_SAFE_INTEGER;
console.log(BigInt(max) + BigInt(max));
```

执行下面的代码，定义存储唯一值的集合数据类型变量。

```javascript
let set = new Set([233, 666, 1.23, 233]);
console.log(set);
console.log(set.size);
for (let v of set) {
	console.log(v);
}
set.add('Hello World!');
console.log(set);
set.delete(666);
console.log(set);
console.log(set.has(233));
set.clear();
console.log(set);
```

执行下面的代码，定义存储键值对的映射数据类型变量。

```javascript
let map = new Map();
map.set('name', 'Tom');
map.set('func', () => 'Hello World!');
let obj = {
    foo: 'bar'
};
map.set(obj, [233, 666, 1.23]);
console.log(map);
console.log(map.size);
for (let v of map) {
    console.log(v);
}
console.log(map.get('name'));
console.log(map.get('func')());
console.log(map.get(obj));
map.delete('name');
console.log(map);
map.clear();
console.log(map);
```

Symbol 是 JavaScript 中一种不可变且唯一的数据类型，通常用来创建独一无二的键。

```javascript
let symbol1 = Symbol();
console.log(symbol1, typeof symbol1);
let symbol2 = Symbol('Hello');
console.log(symbol2, typeof symbol2);
let symbol3 = Symbol('JavaScript');
console.log(symbol3, typeof symbol3);
```
