# 对象使用

执行下面的代码，声明对象并访问其属性。

```javascript
let obj = {
    name: 'Tom',
    age: 18,
    'main-title': 'Hello!',
    'sub title': 'Javascript!'
};
console.log(obj);
console.log(obj.name);
console.log(obj.age);
console.log(obj['main-title']);
console.log(obj['sub title']);
```

执行下面的代码，判断属性是否存在于对象中。

```javascript
let obj = {
    name: 'Tom',
    age: 18
};
console.log('name' in obj);
console.log('age' in obj);
console.log('main-title' in obj);
```

执行下面的代码，删除对象中的指定属性。

```javascript
let obj = {
    name: 'Tom',
    age: 18
};
console.log(obj);
delete obj.name;
console.log(obj);
console.log(obj.name);
```

执行下面的代码，声明对象并访问其方法。

```javascript
let obj = {
    func: () => 'Hello World!'
};
console.log(obj);
console.log(obj.func());
```

执行下面的代码，遍历对象中的所有属性与方法。

```javascript
let obj = {
    name: 'Tom',
    age: 18,
    func: () => 'Hello World!'
};
console.log(Object.keys(obj));
console.log(Object.values(obj));
console.log(Object.entries(obj));
for (let k in obj) {
    console.log(k, obj[k]);
}
```

执行下面的代码，使用 Symbol 作为对象属性名或方法名。

```javascript
let name = Symbol();
let age = Symbol();
let func = Symbol();
let obj = {
    [name]: 'Tom',
    [age]: 18,
    [func]: () => 'Hello World!'
};
console.log(obj);
console.log(obj[name]);
console.log(obj[age]);
console.log(obj[func]());
```

执行下面的代码，使用解构语法为变量赋值。

```javascript
let obj = {
	name: 'Tom',
	age: 18
};
let { name, age } = obj;
console.log(name);
console.log(age);
```

构造函数是一种特殊的函数，主要用来初始化对象。

```javascript
function Person(name, age) {
    this.name = name;
    this.age = age;
}
let Tom = new Person('Tom', 20);
let Jerry = new Person('Jerry', 18);
console.log(Tom);
console.log(Jerry);
```

构造函数的属性和方法被称为静态属性和静态方法。

```javascript
function Person() {}
Person.username = 'anonymous';
Person.func = () => 'Hello World!';
console.log(Person.username);
console.log(Person.func());
```

在 JavaScript 中所有构造函数中都存在一个原型对象。

```javascript
function Person(name, age) {
    this.name = name;
    this.age = age;
}
Person.prototype.func = () => console.log('Hello World!');
let Tom = new Person('Tom', 20);
let Jerry = new Person('Jerry', 18);
console.log(Tom);
Tom.func();
console.log(Jerry);
Jerry.func()
```

执行下面的代码，用定义类的方式来声明对象。

```javascript
class Person {
	name = 'anonymous';
	age = 18;
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
}
let Tom = new Person('tom', 20);
console.log(Tom);
```

执行下面的代码，实现类的继承。

```javascript
class Person {
    name = 'anonymous';
    age = 18;
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
    func() {
        return [this.name, this.age];
    }
}
class SmartPerson extends Person {
    gender;
    constructor(name, age, gender) {
        super(name, age);
        this.gender = gender;
    }
    func() {
        return [this.name, this.age, this.gender];
    }
}
let Tom = new Person('tom', 20);
console.log(Tom);
console.log(Tom.func());
let Jerry = new SmartPerson('jerry', 18, 'female');
console.log(Jerry);
console.log(Jerry.func());
```

执行下面的代码，定义并访问对象中的私有属性。

```javascript
class Person {
	name = 'anonymous';
	#age = 18;
    constructor(name, age) {
        this.name = name;
        this.#age = age;
    }
    func() {
        return this.#age;
    }
}
let Tom = new Person('tom', 20);
console.log(Tom);
console.log(Tom.name);
console.log(Tom.func());
```

执行下面的代码，自定义类的读取和修改行为。

```javascript
class Person {
    #name = 'anonymous';
    get name() {
        console.log('get');
        return this.#name;
    }
    set name(name) {
        console.log('set');
        this.#name = name;
    }
}
let Tom = new Person();
console.log(Tom.name);
Tom.name = 'tom';
console.log(Tom.name);
```

执行下面的代码，使用可选链操作符访问指定变量。

```javascript
let person = {
    name: 'tom',
    age: 20,
    obj: {
        foo: 'bar',
        func: () => 'Hello World!'
    }
};
console.log(person?.obj?.foo);
console.log(person?.obj?.bar);
console.log(person?.obj?.func());
```

执行下面的代码，将不同对象中的属性合并。

```javascript
let obj1 = {
    name: 'Tom',
    age: 20,
    func: () => 'Hello World!'
};
let obj2 = {
    name: 'Jerry',
    age: 18
};
let obj3 = {
    name: 'Spike',
    age: 22,
    gender: 'male'
};
console.log(Object.assign(obj1, obj2, obj3));
console.log(obj1);
console.log(obj2);
console.log(obj3);
```
